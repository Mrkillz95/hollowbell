# 1.7.0: the bell glass set in 3D on an armor stand (front, side, back, close), and the power icon by the hotbar
world flat
gui off
render 6
sim 20
cmd time set 6000
cmd gamerule doDaylightCycle false
cmd gamerule doMobSpawning false
cmd tp @p 0.5 -60 -2.2 0 12
wait 20
cmd summon armor_stand 0.5 -60 0.5 {Rotation:[180f,0f],ShowArms:1b,NoBasePlate:1b,Pose:{RightArm:[-15f,0f,10f],LeftArm:[-10f,0f,-10f]},ArmorItems:[{id:"hollowbell:bell_glass_boots",count:1},{id:"hollowbell:bell_glass_leggings",count:1},{id:"hollowbell:bell_glass_chestplate",count:1},{id:"hollowbell:bell_glass_helmet",count:1}],HandItems:[{id:"hollowbell:stinger",count:1},{}]}
wait 30
shot armour_front
wait 17
shot armour_front_b
cmd tp @p 3.0 -60 0.5 90 12
wait 10
shot armour_side
cmd tp @p 0.5 -60 3.0 180 12
wait 10
shot armour_back
cmd tp @p 0.5 -59.2 -1.3 0 30
wait 10
shot armour_close
cmd tp @p -1.6 -59.6 2.2 -140 20
wait 10
shot armour_back_close
gui on
cmd item replace entity @p armor.head with hollowbell:bell_glass_helmet
cmd item replace entity @p armor.chest with hollowbell:bell_glass_chestplate
cmd item replace entity @p armor.legs with hollowbell:bell_glass_leggings
cmd item replace entity @p armor.feet with hollowbell:bell_glass_boots
cmd item replace entity @p weapon.mainhand with hollowbell:stinger
cmd tp @p 0.5 -60 -2.2 0 12
wait 20
shot armour_hud
quit
