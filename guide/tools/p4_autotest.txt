# Part 4 pictures (ADMIN_BRIDGE.md 19): the Presets page with the re-tune presets, a Pocket Giants world with a few
# giants about two blocks tall next to the player, and a Titans giant.
#   set GUIDE_SCRIPT to this file's full path, then withslot.ps1 <guide> runAutotest -Pgiants
wait 60
guiscale 2
wait 10
adminintro seen
adminmode simple
adminscope world
adminbook presets
wait 60
mouse 2 2
wait 25
shot p4_presets
adminbook presets 200
wait 30
mouse 2 2
wait 25
shot p4_presets_all
# the preview of a preset (Pocket Giants), from the open book
adminreq {"do":"preset","name":"pocket_giants"}
wait 40
mouse 2 2
wait 25
shot p4_preview_pocket
close
# Pocket Giants: applied, then one of each summoned at his pocket size round the player
cmd giants admin preset pocket_giants
cmd giants admin confirm
wait 20
cmd time set 6000
# (for the picture: different kinds this close would fight, and they keep still)
cmd giants meetings off world
cmd giants fight off world
cmd giants admin set pitchgut roam stay
cmd giants admin set furrowmaw roam stay
cmd giants admin set cerberus roam stay
cmd giants admin set hollowbell roam stay
cmd giants admin set lanternwillow roam stay
cmd giants admin set wreckback roam stay
cmd giants admin set swarmforge roam stay
cmd giants admin set gloamwick roam stay
cmd tp @s 0 -60 -7 0 14
wait 20
cmd execute positioned 2.5 -60 4 run wreckback summon calm 0.011
cmd execute positioned -2.5 -60 4 run cerberus summon calm 0.011
cmd execute positioned 0 -60 6 run pitchgut summon calm 0.012
cmd execute positioned 5 -60 6 run gloamwick summon calm 0.011
cmd execute positioned -5 -60 6 run lanternwillow summon calm 0.011
cmd execute positioned 7 -60 3 run swarmforge summon calm 0.015
cmd execute positioned -8 -60 3 run furrowmaw summon calm 0.03
cmd execute positioned 0 -60 9 run hollowbell summon calm 0.011
# (after their arrivals: the Furrowmaw rears up out of the ground, the Hollowbell comes down from the sky)
wait 300
gui off
wait 10
shot p4_pocket_world
cmd tp @s 1 -60 -2 5 22
wait 40
shot p4_pocket_close
gui on
adminbook home
wait 60
mouse 2 2
wait 25
shot p4_pocket_home
close
cmd wreckback remove
cmd cerberus remove
cmd pitchgut remove
cmd gloamwick remove
cmd lanternwillow remove
cmd swarmforge remove
cmd furrowmaw remove
cmd hollowbell remove
wait 40
# Titans: applied; a Cerberus summoned at his usual size comes twice as big
cmd giants admin preset titans
cmd giants admin confirm
wait 20
cmd execute positioned 0 -60 260 run cerberus summon calm
wait 120
cmd tp @s 0 -40 -120 0 -18
wait 120
gui off
wait 10
shot p4_titan_cerberus
gui on
adminbook home
wait 60
mouse 2 2
wait 25
shot p4_titans_home
close
cmd cerberus remove
wait 20
quit
