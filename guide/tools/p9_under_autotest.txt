# Part 9: the Underworld pictures alone (as in p9_autotest.txt)
wait 60
guiscale 2
wait 10
adminintro seen
adminmode simple
adminscope world
cmd time set 6000
cmd gamerule doDaylightCycle false
cmd weather clear
cmd gamerule doMobSpawning false
cmd difficulty normal
cmd gamerule sendCommandFeedback false
cmd giants admin setall maxInWorld 8
cmd pitchgut away off
cmd furrowmaw away off
cmd cerberus away off
cmd hollowbell away off
cmd lanternwillow away off
cmd wreckback away off
cmd swarmforge away off
cmd gloamwick away off
wait 20
wait 200
wait 60
wait 25
# worship up close: the ring round a calm Cerberus, each looking up at him, a loaf laid toward him
cmd giants admin set all villagers worship
cmd tp @s 300 -60 -800 0 10
wait 30
cmd cerberus summon calm 0.05
wait 20
cmd giants freeze
cmd execute at @e[type=fire_ice_cerberus:cerberus,limit=1,sort=nearest] run summon villager ~10 ~ ~4
cmd execute at @e[type=fire_ice_cerberus:cerberus,limit=1,sort=nearest] run summon villager ~-10 ~ ~6
cmd execute at @e[type=fire_ice_cerberus:cerberus,limit=1,sort=nearest] run summon villager ~8 ~ ~-9
cmd execute at @e[type=fire_ice_cerberus:cerberus,limit=1,sort=nearest] run summon villager ~-9 ~ ~-8
cmd execute at @e[type=fire_ice_cerberus:cerberus,limit=1,sort=nearest] run summon villager ~12 ~ ~-1
cmd execute at @e[type=fire_ice_cerberus:cerberus,limit=1,sort=nearest] run summon villager ~-12 ~ ~0
cmd gamemode spectator
wait 500
cmd execute at @e[type=villager,limit=1,sort=nearest] run tp @s ~ ~2.5 ~
cmd execute at @e[type=fire_ice_cerberus:cerberus,limit=1,sort=nearest] run tp @s ^ ^4 ^-15
cmd execute as @s at @s run tp @s ~ ~ ~ facing entity @e[type=fire_ice_cerberus:cerberus,limit=1,sort=nearest] feet
wait 20
shot p9_villagers_worship_ring
cmd execute at @e[type=villager,limit=1,sort=nearest] run tp @s ^1.2 ^1.2 ^-2.5
cmd execute as @s at @s run tp @s ~ ~ ~ facing entity @e[type=fire_ice_cerberus:cerberus,limit=1,sort=nearest] eyes
wait 20
shot p9_villagers_worship_close
cmd execute at @e[type=item,limit=1,sort=nearest] run tp @s ~1 ~2.2 ~1.5
cmd execute as @s at @s run tp @s ~ ~ ~ facing entity @e[type=item,limit=1,sort=nearest] feet
wait 20
shot p9_villagers_offering
cmd giants unfreeze
cmd giants admin set all villagers off
cmd kill @e[type=villager]
cmd cerberus kill
cmd gamemode creative
wait 40
# the Underworld: one of each comes near you in the Nether, sized to fit under its roof (seen with night vision, from
# spectator a little off each)
cmd giants admin set all underworld nether
cmd effect give @s night_vision 99999 0 true
cmd execute in minecraft:the_end run tp @s 100 70 0
wait 400
cmd giants admin spectate furrowmaw
wait 80
shot p9_underworld_end_furrowmaw
cmd giants admin spectate stop
cmd gamemode creative
cmd execute in minecraft:the_nether run tp @s 0 70 0
wait 1500
cmd giants admin spectate hollowbell
wait 80
shot p9_underworld_hollowbell
cmd giants admin spectate pitchgut
wait 80
shot p9_underworld_pitchgut
cmd giants admin spectate gloamwick
wait 80
shot p9_underworld_gloamwick
cmd giants admin spectate lanternwillow
wait 80
shot p9_underworld_lanternwillow
cmd giants admin spectate cerberus
wait 80
shot p9_underworld_cerberus
cmd giants admin spectate wreckback
wait 80
shot p9_underworld_wreckback
cmd giants admin spectate swarmforge
wait 80
shot p9_underworld_swarmforge
